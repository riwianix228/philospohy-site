   667→// #E8B84B + 0.5 -> 'rgba(232,184,75,0.5)'

   669→  const n = parseInt(hex.slice(1), 16);
   670→  return `rgba(${(n>>16)&255},${(n>>8)&255},${n&255},${a})`;

   691→/* --- Свечения: рисуем спрайтами (offscreen canvas) — быстро --- */
   692→function makeGlow(color){
   693→  const s = 96, c = document.createElement('canvas');
   694→  c.width = c.height = s;
   695→  const g = c.getContext('2d');
   696→  const grd = g.createRadialGradient(s/2, s/2, 0, s/2, s/2, s/2);
   697→  grd.addColorStop(0.00, hexToRgba(color, .85));
   698→  grd.addColorStop(0.22, hexToRgba(color, .32));
   699→  grd.addColorStop(0.55, hexToRgba(color, .08));
   700→  grd.addColorStop(1.00, 'rgba(0,0,0,0)');
   701→  g.fillStyle = grd;
   702→  g.fillRect(0, 0, s, s);
   703→  return c;

   719→function initParticles(){

   722→  for (let i = 0; i < total; i++){

   732→      radF: 0.1 + Math.sqrt(Math.random()) * 0.9,        // радиус орбиты (доля)
   733→      orbSpeed: rand(0.08, 0.3) * (Math.random() < .5 ? -1 : 1),

   761→/* --- Шаг физики частиц --- */
   762→function stepParticles(){
   763→  for (const p of particles){
   764→    p.fx += p.vx; p.fy += p.vy;                       // свободный дрейф
   765→    if (p.fx < -20) p.fx = W + 20; if (p.fx > W + 20) p.fx = -20;
   766→    if (p.fy < -20) p.fy = H + 20; if (p.fy > H + 20) p.fy = -20;
   767→    if (p.node >= 0){
   768→      const n = nodes[p.node];
   769→      const ang = p.baseAng + time * p.orbSpeed;      // медленная орбита
   770→      const rad = p.radF * clusterR();
   771→      const tx = n.px + Math.cos(ang) * rad;
   772→      const ty = n.py + Math.sin(ang) * rad * 0.82;   // слегка сплюснуто
   773→      p.x = lerp(p.fx, tx, p.assemble);               // assemble твинит GSAP
   774→      p.y = lerp(p.fy, ty, p.assemble);
   775→    } else { p.x = p.fx; p.y = p.fy; }
   776→  }
   777→}

   780→function draw(){

   785→  for (const s of stars){

   787→    ctx.fillStyle = `rgba(210,218,245,${a})`;
   788→    ctx.beginPath(); ctx.arc(s.x * W, s.y * H, s.r, 0, TAU); ctx.fill();

   795→  ctx.translate(W / 2, H / 2);
   796→  ctx.scale(cam.z, cam.z);
   797→  ctx.translate(-(cam.x + cam.px * framePar), -(cam.y + cam.py * framePar));

   799→  drawEdges();

   801→  drawParticles();
   802→  drawNodes();

   808→function drawEdges(){
   809→  if (worldState.edge <= 0.01) return;
   810→  ctx.lineWidth = 1 / cam.z;                        // всегда волосяная линия
   811→  for (const [i, j] of EDGES){

   814→    g.addColorStop(0, hexToRgba(a.color, 0.36 * worldState.edge * framePar));
   815→    g.addColorStop(1, hexToRgba(b.color, 0.36 * worldState.edge * framePar));
   816→    ctx.strokeStyle = g;
   817→    ctx.beginPath(); ctx.moveTo(a.px, a.py); ctx.lineTo(b.px, b.py); ctx.stroke();
   818→  }

   822→const LINK_DIST = 100, LINK_DIST2 = LINK_DIST * LINK_DIST;
   823→function drawParticleLinks(){
   824→  if (worldState.edge <= 0.01 || cam.z > 2.5) return;   // при зуме пропускаем
   825→  ctx.lineWidth = 1 / cam.z;
   826→  for (let i = 0; i < particles.length; i++){
   827→    const p = particles[i];
   828→    for (let j = i + 1; j < particles.length; j++){
   829→      const q = particles[j];
   830→      const dx = p.x - q.x; if (dx > LINK_DIST || dx < -LINK_DIST) continue;
   831→      const dy = p.y - q.y; if (dy > LINK_DIST || dy < -LINK_DIST) continue;
   832→      const d2 = dx * dx + dy * dy;
   833→      if (d2 < LINK_DIST2){
   834→        const a = (1 - Math.sqrt(d2) / LINK_DIST) * 0.12 * worldState.edge * framePar;
   835→        ctx.strokeStyle = `rgba(154,166,220,${a})`;
   836→        ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo(q.x, q.y); ctx.stroke();
   837→      }
   838→    }
   839→  }
   840→}

   852→function drawNodes(){

   858→
   859→    // ореол (спрайт свечения цвета философа)

   863→
   864→    // ядро
   865→    ctx.globalAlpha = worldState.node;

   868→
   869→    // тонкое кольцо
   870→    ctx.strokeStyle = hexToRgba(n.color, (n.hover ? 0.95 : 0.45) * worldState.node);
   871→    ctx.lineWidth = (n.hover ? 1.5 : 1) / cam.z;
   872→    ctx.beginPath(); ctx.arc(n.px, n.py, cr * 0.62, 0, TAU); ctx.stroke();

   874→    // подпись при наведении
   875→    const target = (n.hover && introActive && !entering && cam.z < 1.2) ? 1 : 0;
   876→    n.labelA = lerp(n.labelA, target, 0.12);

   880→      ctx.font = '500 12px Inter, sans-serif';

   882→      ctx.fillText(n.name.toUpperCase(), n.px, n.py + cr * 0.62 + 24);
   883→      try { ctx.letterSpacing = '0px'; } catch(e){}
   884→    }
   885→  }
   886→  ctx.globalAlpha = 1;
   887→}

  1003→/* ------------------------------------------------------------
  1004→   5. ЗВУК (Web Audio, генерируется на лету — без файлов)
  1005→   Дрон-фон + мягкие бипы + «вжух» на переходе.
  1006→   ------------------------------------------------------------ */
  1007→const Sound = {
  1008→  ctx: null, master: null, noiseBuf: null, muted: true,
  1009→  ensure(){
  1010→    if (this.ctx) return true;
  1011→    try{
  1012→      const AC = window.AudioContext || window.webkitAudioContext;
  1013→      if (!AC) return false;
  1014→      this.ctx = new AC();
  1015→      this.master = this.ctx.createGain();
  1016→      this.master.gain.value = 0;
  1017→      this.master.connect(this.ctx.destination);
  1018→      // фильтр + три расстроенных осциллятора = космический дрон
  1019→      const filt = this.ctx.createBiquadFilter();
  1020→      filt.type = 'lowpass'; filt.frequency.value = 240; filt.Q.value = 0.6;
  1021→      filt.connect(this.master);
  1022→      [[55, 'sine', 0.5], [55.6, 'sine', 0.5], [110.4, 'triangle', 0.12]].forEach(([f, t, v]) => {
  1023→        const o = this.ctx.createOscillator(); o.type = t; o.frequency.value = f;
  1024→        const g = this.ctx.createGain(); g.gain.value = v;
  1025→        o.connect(g); g.connect(filt); o.start();
  1026→      });
  1027→      // медленный LFO дышит фильтром
  1028→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.06;
  1029→      const lg = this.ctx.createGain(); lg.gain.value = 110;
  1030→      lfo.connect(lg); lg.connect(filt.frequency); lfo.start();
  1031→      // тихий шумовой «воздух» (он же источник для whoosh)
  1032→      const nb = this.ctx.createBuffer(1, this.ctx.sampleRate * 2, this.ctx.sampleRate);
  1033→      const d = nb.getChannelData(0);
  1034→      for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
  1035→      this.noiseBuf = nb;
  1036→      const ns = this.ctx.createBufferSource(); ns.buffer = nb; ns.loop = true;
  1037→      const nf = this.ctx.createBiquadFilter(); nf.type = 'bandpass';
  1038→      nf.frequency.value = 420; nf.Q.value = 0.5;
  1039→      const ng = this.ctx.createGain(); ng.gain.value = 0.015;
  1040→      ns.connect(nf); nf.connect(ng); ng.connect(this.master); ns.start();
  1041→      return true;
  1042→    } catch(e){ return false; }
  1043→  },
  1044→  toggle(){
  1045→    if (!this.ensure()) return this.muted;
  1046→    if (this.ctx.resume) this.ctx.resume();
  1047→    this.muted = !this.muted;
  1048→    const t = this.ctx.currentTime;
  1049→    this.master.gain.cancelScheduledValues(t);
  1050→    this.master.gain.linearRampToValueAtTime(this.muted ? 0 : 0.055, t + 0.5);
  1051→    const btn = document.getElementById('mute-btn');
  1052→    btn.classList.toggle('unmuted', !this.muted);
  1053→    btn.setAttribute('aria-pressed', String(!this.muted));
  1054→    return this.muted;
  1055→  },
  1056→  blip(){                                  // тёплый короткий клик
  1057→    if (!this.ctx || this.muted) return;
  1058→    const t = this.ctx.currentTime;
  1059→    const o = this.ctx.createOscillator(); o.type = 'sine';
  1060→    o.frequency.setValueAtTime(540, t);
  1061→    o.frequency.exponentialRampToValueAtTime(320, t + 0.16);
  1062→    const g = this.ctx.createGain();
  1063→    g.gain.setValueAtTime(0.0001, t);
  1064→    g.gain.exponentialRampToValueAtTime(0.12, t + 0.02);
  1065→    g.gain.exponentialRampToValueAtTime(0.0001, t + 0.22);
  1066→    o.connect(g); g.connect(this.ctx.destination);
  1067→    o.start(t); o.stop(t + 0.25);
  1068→  },
  1069→  whoosh(){                                // шумовой подъём для zoom-перехода
  1070→    if (!this.ctx || this.muted) return;
  1071→    const t = this.ctx.currentTime;
  1072→    const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf;
  1073→    const f = this.ctx.createBiquadFilter(); f.type = 'bandpass'; f.Q.value = 1.2;
  1074→    f.frequency.setValueAtTime(180, t);
  1075→    f.frequency.exponentialRampToValueAtTime(2400, t + 1.5);
  1076→    const g = this.ctx.createGain();
  1077→    g.gain.setValueAtTime(0.0001, t);

  1464→// ветви лавра: листья по дуге — без единой картинки
  1465→function laurelSVG(){
  1466→  const cx = 75, cy = 95, R = 55;
  1467→  let s = '';
  1468→  const branch = (a0, a1) => {
  1469→    for (let i = 0; i <= 6; i++){
  1470→      const t  = (a0 + (a1 - a0) * i / 6) * Math.PI / 180;
  1471→      const x  = cx + R * Math.cos(t), y = cy + R * Math.sin(t);
  1472→      const nx = Math.cos(t), ny = Math.sin(t);   // наружу от центра
  1473→      const tx = -ny, ty = nx;                    // вдоль ветви
  1474→      const L  = 11 - i;
  1475→      s += `<line x1="${x.toFixed(1)}" y1="${y.toFixed(1)}" x2="${(x + tx * L * .95).toFixed(1)}" y2="${(y + ty * L * .95).toFixed(1)}"/>`;
  1476→      s += `<line x1="${x.toFixed(1)}" y1="${y.toFixed(1)}" x2="${(x + nx * L * .55 + tx * L * .35).toFixed(1)}" y2="${(y + ny * L * .55 + ty * L * .35).toFixed(1)}"/>`;
  1477→    }
  1478→  };
  1479→  branch(146, 214); branch(34, -34);
  1480→  return s;
  1481→}

  1516→  <g stroke="${col}" fill="none" stroke-width="1.3" stroke-linecap="round" opacity=".15">${glyph}</g>
  1517→  ${starDots}
