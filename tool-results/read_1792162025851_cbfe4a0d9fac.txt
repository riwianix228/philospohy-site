   842→/* ------------------------------------------------------------
   843→   5. ЗВУК (Web Audio, генерируется на лету — без файлов)
   844→   Дрон-фон + мягкие бипы + «вжух» на переходе.
   845→   ------------------------------------------------------------ */
   846→const Sound = {
   847→  ctx: null, master: null, noiseBuf: null, muted: true,
   848→  ensure(){
   849→    if (this.ctx) return true;
   850→    try{
   851→      const AC = window.AudioContext || window.webkitAudioContext;
   852→      if (!AC) return false;
   853→      this.ctx = new AC();
   854→      this.master = this.ctx.createGain();
   855→      this.master.gain.value = 0;
   856→      this.master.connect(this.ctx.destination);
   857→      // фильтр + три расстроенных осциллятора = космический дрон
   858→      const filt = this.ctx.createBiquadFilter();
   859→      filt.type = 'lowpass'; filt.frequency.value = 240; filt.Q.value = 0.6;
   860→      filt.connect(this.master);
   861→      [[55, 'sine', 0.5], [55.6, 'sine', 0.5], [110.4, 'triangle', 0.12]].forEach(([f, t, v]) => {
   862→        const o = this.ctx.createOscillator(); o.type = t; o.frequency.value = f;
   863→        const g = this.ctx.createGain(); g.gain.value = v;
   864→        o.connect(g); g.connect(filt); o.start();
   865→      });
   866→      // медленный LFO дышит фильтром
   867→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.06;
   868→      const lg = this.ctx.createGain(); lg.gain.value = 110;
   869→      lfo.connect(lg); lg.connect(filt.frequency); lfo.start();
   870→      // тихий шумовой «воздух» (он же источник для whoosh)
   871→      const nb = this.ctx.createBuffer(1, this.ctx.sampleRate * 2, this.ctx.sampleRate);
   872→      const d = nb.getChannelData(0);
   873→      for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
   874→      this.noiseBuf = nb;
   875→      const ns = this.ctx.createBufferSource(); ns.buffer = nb; ns.loop = true;
   876→      const nf = this.ctx.createBiquadFilter(); nf.type = 'bandpass';
   877→      nf.frequency.value = 420; nf.Q.value = 0.5;
   878→      const ng = this.ctx.createGain(); ng.gain.value = 0.015;
   879→      ns.connect(nf); nf.connect(ng); ng.connect(this.master); ns.start();
   880→      return true;
   881→    } catch(e){ return false; }
   882→  },
   883→  toggle(){
   884→    if (!this.ensure()) return this.muted;
   885→    if (this.ctx.resume) this.ctx.resume();
   886→    this.muted = !this.muted;
   887→    const t = this.ctx.currentTime;
   888→    this.master.gain.cancelScheduledValues(t);
   889→    this.master.gain.linearRampToValueAtTime(this.muted ? 0 : 0.055, t + 0.5);
   890→    const btn = document.getElementById('mute-btn');
   891→    btn.classList.toggle('unmuted', !this.muted);
   892→    btn.setAttribute('aria-pressed', String(!this.muted));
   893→    return this.muted;
   894→  },
   895→  blip(){                                  // тёплый короткий клик
   896→    if (!this.ctx || this.muted) return;
   897→    const t = this.ctx.currentTime;
   898→    const o = this.ctx.createOscillator(); o.type = 'sine';
   899→    o.frequency.setValueAtTime(540, t);
   900→    o.frequency.exponentialRampToValueAtTime(320, t + 0.16);
   901→    const g = this.ctx.createGain();
   902→    g.gain.setValueAtTime(0.0001, t);
   903→    g.gain.exponentialRampToValueAtTime(0.12, t + 0.02);
   904→    g.gain.exponentialRampToValueAtTime(0.0001, t + 0.22);
   905→    o.connect(g); g.connect(this.ctx.destination);
   906→    o.start(t); o.stop(t + 0.25);
   907→  },
   908→  whoosh(){                                // шумовой подъём для zoom-перехода
   909→    if (!this.ctx || this.muted) return;
   910→    const t = this.ctx.currentTime;
   911→    const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf;
   912→    const f = this.ctx.createBiquadFilter(); f.type = 'bandpass'; f.Q.value = 1.2;
   913→    f.frequency.setValueAtTime(180, t);
   914→    f.frequency.exponentialRampToValueAtTime(2400, t + 1.5);
   915→    const g = this.ctx.createGain();
   916→    g.gain.setValueAtTime(0.0001, t);
   917→    g.gain.exponentialRampToValueAtTime(0.28, t + 1.2);
   918→    g.gain.exponentialRampToValueAtTime(0.0001, t + 1.7);
   919→    src.connect(f); f.connect(g); g.connect(this.ctx.destination);
   920→    src.start(t); src.stop(t + 1.8);
   921→  }
   922→};
