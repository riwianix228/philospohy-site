   664→const lerp = (a, b, t) => a + (b - a) * t;

   667→// #E8B84B + 0.5 -> 'rgba(232,184,75,0.5)'
   668→function hexToRgba(hex, a = 1){
   669→  const n = parseInt(hex.slice(1), 16);
   670→  return `rgba(${(n>>16)&255},${(n>>8)&255},${n&255},${a})`;
   671→}

   687→let introActive = false;

   691→/* --- Свечения: рисуем спрайтами (offscreen canvas) — быстро --- */
   692→function makeGlow(color){
   693→  const s = 96, c = document.createElement('canvas');
   694→  c.width = c.height = s;
   695→  const g = c.getContext('2d');
   696→  const grd = g.createRadialGradient(s/2, s/2, 0, s/2, s/2, s/2);
   697→  grd.addColorStop(0.00, hexToRgba(color, .85));
   698→  grd.addColorStop(0.22, hexToRgba(color, .32));
   699→  grd.addColorStop(0.55, hexToRgba(color, .08));
   700→  grd.addColorStop(1.00, 'rgba(0,0,0,0)');
   701→  g.fillStyle = grd;
   702→  g.fillRect(0, 0, s, s);
   703→  return c;
   704→}
   705→const sprites = {};
   706→PHILOSOPHERS.forEach(p => sprites[p.id] = makeGlow(p.color));

   709→/* --- Узлы-философы --- */
   710→const nodes = PHILOSOPHERS.map((p, i) => ({
   711→  ...p, i, px: 0, py: 0,          // px/py — экранные координаты
   712→  hover: false, labelA: 0,        // плавное появление подписи
   713→  pulse: rand(0, TAU),            // фаза пульсации

   715→function clusterR(){ return Math.max(22, Math.min(44, Math.min(W, H) * 0.045)); }

   717→/* --- Частицы --- */
   718→const particles = [];
   719→function initParticles(){
   720→  particles.length = 0;
   721→  const total = Math.round(Math.min(240, Math.max(90, (W * H) / 6000)));
   722→  for (let i = 0; i < total; i++){
   723→    const colored = Math.random() < 0.5;                 // половина — «цветные»
   724→    const node = colored ? Math.floor(rand(0, 6)) : -1;  // летят к своему узлу
   725→    particles.push({
   726→      fx: rand(0, W), fy: rand(0, H),                    // свободный дрейф
   727→      vx: rand(-0.12, 0.12), vy: rand(-0.12, 0.12),
   728→      r: rand(0.6, 1.9),
   729→      sprite: node >= 0 ? sprites[PHILOSOPHERS[node].id] : spriteNeutral,
   730→      node,
   731→      baseAng: rand(0, TAU),                             // орбита вокруг узла
   732→      radF: 0.1 + Math.sqrt(Math.random()) * 0.9,        // радиус орбиты (доля)
   733→      orbSpeed: rand(0.08, 0.3) * (Math.random() < .5 ? -1 : 1),
   734→      tw: rand(0.4, 1.4), ph: rand(0, TAU),              // мерцание
   735→      assemble: REDUCED ? 1 : 0,                         // 0 = хаос, 1 = в узле
   736→      x: 0, y: 0,
   737→    });
   738→  }
   739→}

   743→function initStars(){

   748→  }

   760→function resize(){

   762→  W = window.innerWidth; H = window.innerHeight;
   763→  canvas.width = Math.round(W * DPR); canvas.height = Math.round(H * DPR);
   764→  ctx.setTransform(DPR, 0, 0, DPR, 0, 0);

   766→  if (cam.z === 1){ cam.x = W / 2; cam.y = H / 2; }
   767→  // холст сетки — те же размеры и плотность пикселей

   772→/* --- Шаг физики частиц --- */
   773→function stepParticles(){
   774→  for (const p of particles){
   775→    p.fx += p.vx; p.fy += p.vy;                       // свободный дрейф
   776→    if (p.fx < -20) p.fx = W + 20; if (p.fx > W + 20) p.fx = -20;
   777→    if (p.fy < -20) p.fy = H + 20; if (p.fy > H + 20) p.fy = -20;
   778→    if (p.node >= 0){
   779→      const n = nodes[p.node];
   780→      const ang = p.baseAng + time * p.orbSpeed;      // медленная орбита
   781→      const rad = p.radF * clusterR();
   782→      const tx = n.px + Math.cos(ang) * rad;
   783→      const ty = n.py + Math.sin(ang) * rad * 0.82;   // слегка сплюснуто
   784→      p.x = lerp(p.fx, tx, p.assemble);               // assemble твинит GSAP
   785→      p.y = lerp(p.fy, ty, p.assemble);
   786→    } else { p.x = p.fx; p.y = p.fy; }
   787→  }
   788→}

   790→/* --- Рендер --- */
   791→function draw(){
   792→  ctx.fillStyle = '#0a0a0f';
   793→  ctx.fillRect(0, 0, W, H);
   794→
   795→  // звёзды фона
   796→  for (const s of stars){
   797→    const a = 0.05 + 0.2 * (0.5 + 0.5 * Math.sin(time * s.tw + s.ph));
   798→    ctx.fillStyle = `rgba(210,218,245,${a})`;
   799→    ctx.beginPath(); ctx.arc(s.x * W, s.y * H, s.r, 0, TAU); ctx.fill();
   800→  }
   801→
   802→  // параллакс гаснет при зуме
   803→  framePar = 1 - Math.min(1, (cam.z - 1) / 3);
   804→
   805→  ctx.save();
   806→  ctx.translate(W / 2, H / 2);
   807→  ctx.scale(cam.z, cam.z);
   808→  ctx.translate(-(cam.x + cam.px * framePar), -(cam.y + cam.py * framePar));
   809→
   810→  drawEdges();
   811→  drawParticleLinks();
   812→  drawParticles();
   813→  drawNodes();
   814→
   815→  ctx.restore();
   816→}

   819→function startGrid3d(color){
   820→  if (REDUCED || !HAS_GSAP) return;            // в статичном режиме сетка не нужна
   821→  grid3d.color = color;
   822→  grid3d.active = true;

   824→  gsap.killTweensOf(grid3d);

   826→  // линии разгоняются по мере влёта камеры в узел
   827→  gsap.fromTo(grid3d, { speed: .45 }, { speed: 3.6, duration: 1.9, ease: 'power2.in' });
   828→}
   829→function stopGrid3d(){
   830→  if (!grid3d.active) return;
   831→  gsap.killTweensOf(grid3d);
   832→  gsap.to(grid3d, {
   833→    alpha: 0, duration: .4, ease: 'power1.out',
   834→    onComplete: () => { grid3d.active = false; gctx.clearRect(0, 0, W, H); }
   835→  });

   837→function drawGrid3d(dt){
   838→  if (!grid3d.active || grid3d.alpha <= 0.005) return;
   839→  gctx.clearRect(0, 0, W, H);
   840→  gctx.save();
   841→  gctx.globalAlpha = grid3d.alpha;
   842→  const hy = H * 0.46;                          // линия горизонта
   843→  // тёплое свечение неба у горизонта
   844→  const halo = gctx.createLinearGradient(0, hy - 70, 0, hy + 120);
   845→  halo.addColorStop(0, 'rgba(0,0,0,0)');
   846→  halo.addColorStop(.55, hexToRgba(grid3d.color, .16 * grid3d.alpha));
   847→  halo.addColorStop(1, 'rgba(0,0,0,0)');
   848→  gctx.fillStyle = halo;
   849→  gctx.fillRect(0, hy - 70, W, 190);
   850→  // сама линия горизонта — ярче к центру
   851→  const hl = gctx.createLinearGradient(0, 0, W, 0);
   852→  hl.addColorStop(0, hexToRgba(grid3d.color, 0));
   853→  hl.addColorStop(.5, hexToRgba(grid3d.color, .9 * grid3d.alpha));
   854→  hl.addColorStop(1, hexToRgba(grid3d.color, 0));
   855→  gctx.strokeStyle = hl; gctx.lineWidth = 1.4;
   856→  gctx.beginPath(); gctx.moveTo(0, hy); gctx.lineTo(W, hy); gctx.stroke();
   857→  // вертикальные «рельсы», сходящиеся к точке схода на горизонте
   858→  const NV = 13, spreadB = W * 0.17, spreadT = W * 0.012;
   859→  for (let i = -NV; i <= NV; i++){
   860→    const g2 = gctx.createLinearGradient(0, hy, 0, H);
   861→    g2.addColorStop(0, hexToRgba(grid3d.color, 0));
   862→    g2.addColorStop(1, hexToRgba(grid3d.color, .55 * grid3d.alpha));
   863→    gctx.strokeStyle = g2; gctx.lineWidth = 1;
   864→    gctx.beginPath();
   865→    gctx.moveTo(W / 2 + i * spreadT, hy);
   866→    gctx.lineTo(W / 2 + i * spreadB, H);
   867→    gctx.stroke();
   868→  }
   869→  // бегущие горизонтальные линии: к горизонту — плотнее (перспектива)
   870→  grid3d.off = (grid3d.off + dt * grid3d.speed) % 1;
   871→  const ROWS = 20;
   872→  for (let k = 0; k < ROWS; k++){
   873→    const p = (k + grid3d.off) / ROWS;
   874→    if (p <= 0.001) continue;
   875→    const y = hy + (H - hy) * Math.pow(p, 2.5);
   876→    gctx.strokeStyle = hexToRgba(grid3d.color, Math.pow(p, 1.7) * .6 * grid3d.alpha);
   877→    gctx.lineWidth = p > .6 ? 1.4 : 1;
   878→    gctx.beginPath(); gctx.moveTo(0, y); gctx.lineTo(W, y); gctx.stroke();
   879→  }
   880→  gctx.restore();
   881→}

   883→// линии-связи между шестью узлами
   884→function drawEdges(){
   885→  if (worldState.edge <= 0.01) return;
   886→  ctx.lineWidth = 1 / cam.z;                        // всегда волосяная линия
   887→  for (const [i, j] of EDGES){
   888→    const a = nodes[i], b = nodes[j];
   889→    const g = ctx.createLinearGradient(a.px, a.py, b.px, b.py);
   890→    g.addColorStop(0, hexToRgba(a.color, 0.36 * worldState.edge * framePar));
   891→    g.addColorStop(1, hexToRgba(b.color, 0.36 * worldState.edge * framePar));
   892→    ctx.strokeStyle = g;
   893→    ctx.beginPath(); ctx.moveTo(a.px, a.py); ctx.lineTo(b.px, b.py); ctx.stroke();
   894→  }
   895→}

   897→// «созвездие»: тонкие линии между близкими частицами
   898→const LINK_DIST = 100, LINK_DIST2 = LINK_DIST * LINK_DIST;
   899→function drawParticleLinks(){
   900→  if (worldState.edge <= 0.01 || cam.z > 2.5) return;   // при зуме пропускаем
   901→  ctx.lineWidth = 1 / cam.z;
   902→  for (let i = 0; i < particles.length; i++){
   903→    const p = particles[i];
   904→    for (let j = i + 1; j < particles.length; j++){
   905→      const q = particles[j];
   906→      const dx = p.x - q.x; if (dx > LINK_DIST || dx < -LINK_DIST) continue;
   907→      const dy = p.y - q.y; if (dy > LINK_DIST || dy < -LINK_DIST) continue;
   908→      const d2 = dx * dx + dy * dy;
   909→      if (d2 < LINK_DIST2){
   910→        const a = (1 - Math.sqrt(d2) / LINK_DIST) * 0.12 * worldState.edge * framePar;
   911→        ctx.strokeStyle = `rgba(154,166,220,${a})`;
   912→        ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo(q.x, q.y); ctx.stroke();
   913→      }
   914→    }
   915→  }
   916→}

   918→function drawParticles(){
   919→  for (const p of particles){
   920→    const twinkle = 0.65 + 0.35 * Math.sin(time * p.tw + p.ph);
   921→    const s = p.r * 7;                                 // размер спрайта
   922→    ctx.globalAlpha = twinkle * (0.35 + 0.65 * worldState.node);
   923→    ctx.drawImage(p.sprite, p.x - s / 2, p.y - s / 2, s, s);
   924→  }
   925→  ctx.globalAlpha = 1;
   926→}

   928→function drawNodes(){
   929→  if (worldState.node <= 0.01) return;
   930→  const cr = clusterR();
   931→  ctx.textAlign = 'center';
   932→  for (const n of nodes){
   933→    const pulse = REDUCED ? 1 : 1 + 0.07 * Math.sin(time * 1.3 + n.pulse);
   934→
   935→    // ореол (спрайт свечения цвета философа)
   936→    const gs = cr * 3.4 * pulse;
   937→    ctx.globalAlpha = 0.85 * worldState.node;
   938→    ctx.drawImage(sprites[n.id], n.px - gs / 2, n.py - gs / 2, gs, gs);
   939→
   940→    // ядро
   941→    ctx.globalAlpha = worldState.node;
   942→    ctx.fillStyle = '#fff';
   943→    ctx.beginPath(); ctx.arc(n.px, n.py, 3.2, 0, TAU); ctx.fill();
   944→
   945→    // тонкое кольцо
   946→    ctx.strokeStyle = hexToRgba(n.color, (n.hover ? 0.95 : 0.45) * worldState.node);
   947→    ctx.lineWidth = (n.hover ? 1.5 : 1) / cam.z;
   948→    ctx.beginPath(); ctx.arc(n.px, n.py, cr * 0.62, 0, TAU); ctx.stroke();
   949→
   950→    // подпись при наведении
   951→    const target = (n.hover && introActive && !entering && cam.z < 1.2) ? 1 : 0;
   952→    n.labelA = lerp(n.labelA, target, 0.12);
   953→    if (n.labelA > 0.02){
   954→      ctx.globalAlpha = n.labelA * worldState.node;
   955→      ctx.fillStyle = hexToRgba(n.color, 0.95);
   956→      ctx.font = '500 12px Inter, sans-serif';
   957→      try { ctx.letterSpacing = '3px'; } catch(e){}
   958→      ctx.fillText(n.name.toUpperCase(), n.px, n.py + cr * 0.62 + 24);
   959→      try { ctx.letterSpacing = '0px'; } catch(e){}
   960→    }
   961→  }
   962→  ctx.globalAlpha = 1;
   963→}

  1080→/* ------------------------------------------------------------
  1081→   5. ЗВУК (Web Audio, генерируется на лету — без файлов)
  1082→   Дрон-фон + мягкие бипы + «вжух» на переходе.
  1083→   ------------------------------------------------------------ */
  1084→const Sound = {
  1085→  ctx: null, master: null, noiseBuf: null, muted: true,
  1086→  ensure(){
  1087→    if (this.ctx) return true;
  1088→    try{
  1089→      const AC = window.AudioContext || window.webkitAudioContext;
  1090→      if (!AC) return false;
  1091→      this.ctx = new AC();
  1092→      this.master = this.ctx.createGain();
  1093→      this.master.gain.value = 0;
  1094→      this.master.connect(this.ctx.destination);
  1095→      // фильтр + три расстроенных осциллятора = космический дрон
  1096→      const filt = this.ctx.createBiquadFilter();
  1097→      filt.type = 'lowpass'; filt.frequency.value = 240; filt.Q.value = 0.6;
  1098→      filt.connect(this.master);
  1099→      [[55, 'sine', 0.5], [55.6, 'sine', 0.5], [110.4, 'triangle', 0.12]].forEach(([f, t, v]) => {
  1100→        const o = this.ctx.createOscillator(); o.type = t; o.frequency.value = f;
  1101→        const g = this.ctx.createGain(); g.gain.value = v;
  1102→        o.connect(g); g.connect(filt); o.start();
  1103→      });
  1104→      // медленный LFO дышит фильтром
  1105→      const lfo = this.ctx.createOscillator(); lfo.frequency.value = 0.06;
  1106→      const lg = this.ctx.createGain(); lg.gain.value = 110;
  1107→      lfo.connect(lg); lg.connect(filt.frequency); lfo.start();
  1108→      // тихий шумовой «воздух» (он же источник для whoosh)
  1109→      const nb = this.ctx.createBuffer(1, this.ctx.sampleRate * 2, this.ctx.sampleRate);
  1110→      const d = nb.getChannelData(0);
  1111→      for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
  1112→      this.noiseBuf = nb;
  1113→      const ns = this.ctx.createBufferSource(); ns.buffer = nb; ns.loop = true;
  1114→      const nf = this.ctx.createBiquadFilter(); nf.type = 'bandpass';
  1115→      nf.frequency.value = 420; nf.Q.value = 0.5;
  1116→      const ng = this.ctx.createGain(); ng.gain.value = 0.015;
  1117→      ns.connect(nf); nf.connect(ng); ng.connect(this.master); ns.start();
  1118→      return true;
  1119→    } catch(e){ return false; }
  1120→  },
  1121→  toggle(){
  1122→    if (!this.ensure()) return this.muted;
  1123→    if (this.ctx.resume) this.ctx.resume();
  1124→    this.muted = !this.muted;
  1125→    const t = this.ctx.currentTime;
  1126→    this.master.gain.cancelScheduledValues(t);
  1127→    this.master.gain.linearRampToValueAtTime(this.muted ? 0 : 0.055, t + 0.5);
  1128→    const btn = document.getElementById('mute-btn');
  1129→    btn.classList.toggle('unmuted', !this.muted);
  1130→    btn.setAttribute('aria-pressed', String(!this.muted));
  1131→    return this.muted;
  1132→  },
  1133→  blip(){                                  // тёплый короткий клик
  1134→    if (!this.ctx || this.muted) return;
  1135→    const t = this.ctx.currentTime;
  1136→    const o = this.ctx.createOscillator(); o.type = 'sine';
  1137→    o.frequency.setValueAtTime(540, t);
  1138→    o.frequency.exponentialRampToValueAtTime(320, t + 0.16);
  1139→    const g = this.ctx.createGain();
  1140→    g.gain.setValueAtTime(0.0001, t);
  1141→    g.gain.exponentialRampToValueAtTime(0.12, t + 0.02);
  1142→    g.gain.exponentialRampToValueAtTime(0.0001, t + 0.22);
  1143→    o.connect(g); g.connect(this.ctx.destination);
  1144→    o.start(t); o.stop(t + 0.25);
  1145→  },
  1146→  whoosh(){                                // шумовой подъём для zoom-перехода
  1147→    if (!this.ctx || this.muted) return;
  1148→    const t = this.ctx.currentTime;
  1149→    const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf;
  1150→    const f = this.ctx.createBiquadFilter(); f.type = 'bandpass'; f.Q.value = 1.2;
  1151→    f.frequency.setValueAtTime(180, t);
  1152→    f.frequency.exponentialRampToValueAtTime(2400, t + 1.5);
  1153→    const g = this.ctx.createGain();
  1154→    g.gain.setValueAtTime(0.0001, t);
  1155→    g.gain.exponentialRampToValueAtTime(0.28, t + 1.2);
  1156→    g.gain.exponentialRampToValueAtTime(0.0001, t + 1.7);
  1157→    src.connect(f); f.connect(g); g.connect(this.ctx.destination);
  1158→    src.start(t); src.stop(t + 1.8);
  1159→  }
  1160→};

  1575→// Лавровый венок Марка Аврелия: эллиптическая лента вокруг головы + листья — без картинок
  1576→function laurelWreath(col){
  1577→  const cx = 75, cy = 86, rx = 42, ry = 28;       // венок охватывает голову, верх касается волос
  1578→  const a0 = 200, a1 = 340, N = 12;               // дуга над головой 200°…340°
  1579→  const P = (aDeg, k = 1) => {
  1580→    const a = aDeg * Math.PI / 180;
  1581→    return [cx + rx * k * Math.cos(a), cy + ry * k * Math.sin(a)];
  1582→  };
  1583→  let s = '';
  1584→  // лента венка — две дуги (наружная и внутренняя)
  1585→  const [sx1, sy1] = P(a0), [ex1, ey1] = P(a1);
  1586→  s += `<path d="M${sx1.toFixed(1)} ${sy1.toFixed(1)} A${rx} ${ry} 0 0 1 ${ex1.toFixed(1)} ${ey1.toFixed(1)}"/>`;
  1587→  const [sx2, sy2] = P(a0, .87), [ex2, ey2] = P(a1, .87);
  1588→  s += `<path d="M${sx2.toFixed(1)} ${sy2.toFixed(1)} A${(rx * .87).toFixed(1)} ${(ry * .87).toFixed(1)} 0 0 1 ${ex2.toFixed(1)} ${ey2.toFixed(1)}"/>`;
  1589→  // листья-«линзы» вдоль ленты, слегка тонированные цветом философа
  1590→  for (let i = 0; i <= N; i++){
  1591→    const a  = (a0 + (a1 - a0) * i / N) * Math.PI / 180;
  1592→    const px = cx + rx * Math.cos(a), py = cy + ry * Math.sin(a);
  1593→    let tx = -rx * Math.sin(a), ty = ry * Math.cos(a);          // касательная к эллипсу
  1594→    const tl = Math.hypot(tx, ty); tx /= tl; ty /= tl;
  1595→    let nx = Math.cos(a) / rx, ny = Math.sin(a) / ry;           // нормаль наружу
  1596→    const nl = Math.hypot(nx, ny); nx /= nl; ny /= nl;
  1597→    const L = 9, Wd = 3.6;
  1598→    const qx = px + tx * L, qy = py + ty * L;
  1599→    s += `<path fill="${col}" fill-opacity=".14" stroke="none" d="M${(px + nx * 1.5).toFixed(1)} ${(py + ny * 1.5).toFixed(1)} Q${(px + nx * Wd).toFixed(1)} ${(py + ny * Wd).toFixed(1)} ${qx.toFixed(1)} ${qy.toFixed(1)} Q${(qx - nx * Wd * .5).toFixed(1)} ${(qy - ny * Wd * .5).toFixed(1)} ${(px + nx * 1.5).toFixed(1)} ${(py + ny * 1.5).toFixed(1)} Z"/>`;
  1600→    s += `<path d="M${(px + nx * 1.5).toFixed(1)} ${(py + ny * 1.5).toFixed(1)} Q${(px + nx * Wd).toFixed(1)} ${(py + ny * Wd).toFixed(1)} ${qx.toFixed(1)} ${qy.toFixed(1)} Q${(qx - nx * Wd * .5).toFixed(1)} ${(qy - ny * Wd * .5).toFixed(1)} ${(px + nx * 1.5).toFixed(1)} ${(py + ny * 1.5).toFixed(1)} Z"/>`;
  1601→  }
  1602→  return s;
  1603→}
